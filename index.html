Installation Locale

Prérequis : Python 3.10+, (optionnel) NVIDIA GPU + CUDA 11.8+

# 1. Extraire l'archive
cd quantum_os_v5

# 2. Environnement virtuel
python3 -m venv venv
source venv/bin/activate   # Linux/Mac
# venv\Scripts\activate    # Windows

# 3. Dépendances
pip install -r requirements.txt

# 4. (Optionnel) GPU CUDA
pip install cupy-cuda12x

Docker Recommandé

Production-ready avec monitoring intégré (Prometheus, Grafana, Jaeger).

# Single node
docker build -t quantum-os:v5 .
docker run -it --gpus all quantum-os:v5

# Cluster 3 nodes + monitoring
docker-compose up --build

Modes de démarrage

# Demo (test rapide)
python main.py --demo

# Production
python main.py --config config/default.yaml \
               --mode production

# Cluster — Node 0 (leader)
QOS_NODE_ID=node-0 python main.py \
  --cluster --port 50051

# Cluster — Node 1
QOS_NODE_ID=node-1 \
QOS_CLUSTER_SEEDS=node-0:50051 \
python main.py --cluster --port 50052

Soumettre des tâches

import asyncio
from core.kernel import QuantumKernel
from core.task import QuantumTask, TaskPriority

async def main():
    kernel = QuantumKernel()
    
    # Register engines
    from engines.transformer_engine import TransformerEngine
    from engines.quantum_controller import QuantumController
    kernel.register_engine("transformer", TransformerEngine({}))
    kernel.register_engine("quantum",
        QuantumController({"num_qubits": 5}))
    
    await kernel.boot()
    
    task = QuantumTask(
        id="my-task-1",
        type="transformer",
        data=[1.0, 2.0, 3.0, 4.0],
        priority=TaskPriority.HIGH,
    )
    kernel.submit_task(task)
    await kernel.run(num_workers=4, runtime=30)

asyncio.run(main())

Pipeline DAG

from pipeline.dag_executor import (
    DAGExecutor, PipelineNode
)

dag = DAGExecutor()
dag.add_node(PipelineNode(
    id="preprocess",
    task_type="transformer",
    data=my_data
))
dag.add_node(PipelineNode(
    id="quantum_opt",
    task_type="quantum",
    dependencies=["preprocess"]
))
dag.add_node(PipelineNode(
    id="validate",
    task_type="photonic",
    dependencies=["quantum_opt"]
))

result = await dag.execute(my_execute_fn)
print(f"Done: {result.nodes_completed}",
      f"Failed: {result.nodes_failed}")

Fichier principal : config/default.yaml

system:
  name: "QuantumOS-v5"
  mode: production       # production | development | testing
  log_level: INFO

kernel:
  max_workers: 16
  runtime: 0             # 0 = run forever

scheduler:
  learning_rate: 0.001
  gamma: 0.99
  state_dim: 64
  hidden_dim: 256
  entropy_coeff: 0.01

resources:
  cpu_cores: 16
  gpu_devices: 4
  memory_gb: 64
  quantum_qubits: 128
  photonic_channels: 16

gpu:
  enabled: true
  mixed_precision: true
  fallback_cpu: true

cluster:
  enabled: true
  port: 50051
  node_id: node-0
  seeds: []

sandbox:
  enabled: true
  backend: process       # process | cgroup | namespace
  memory_limit_mb: 512
  timeout_seconds: 300
